3. 콘솔 테스트 코드

// Firebase 동기화 취소 테스트
const testData = Array.from({ length: 2000 }, (_, i) => ({
    id: `test-${i}`,
    name: `Sample ${i}`,
    date: new Date().toISOString()
}));

const controller = new AbortController();

// 동기화 시작
window.firestoreDb.batchSave('soil', 2024, testData, { signal: controller.signal });

// 3초 후 취소
setTimeout(() => controller.abort(), 3000);
